== 4. Genetic algorithm

  The sum of the squares of the even elements is a filter, then a map, then an
  aggregate. No top-down hypothesis splits it, so the GA searches pipelines.
    [1,2,3,4] -> 20
    [2] -> 4
    [1,3] -> 0
    [6,1] -> 36
    [] -> 0
  hypothesis tree:
    sum_sq_evens: genetic algorithm, generation 26: filter((x mod 2) < 1) then map((x*x)) then sum_list
  program:
    sum_sq_evens(A, B) :-
        sum_sq_evens_s1(A, C),
        sum_sq_evens_s2(C, D),
        sum_sq_evens_s3(D, B).
    sum_sq_evens_s1([], []).
    sum_sq_evens_s1([A|B], C) :-
        (   sum_sq_evens_s1_test(A)
        ->  C=[A|D]
        ;   C=D
        ),
        sum_sq_evens_s1(B, D).
    sum_sq_evens_s1_test(A) :-
        A mod 2<1.
    sum_sq_evens_s2([], []).
    sum_sq_evens_s2([A|B], [C|D]) :-
        sum_sq_evens_s2_el(A, C),
        sum_sq_evens_s2(B, D).
    sum_sq_evens_s2_el(A, B) :-
        B is A*A.
    sum_sq_evens_s3(A, B) :-
        sum_list(A, B).
  reference program:
    ref_sum_sq_evens(A, B) :-
        evens(A, C),
        squares(C, D),
        sum_list(D, B).
    evens([], []).
    evens([A|B], C) :-
        (   A mod 2=:=0
        ->  C=[A|D]
        ;   C=D
        ),
        evens(B, D).
    squares([], []).
    squares([A|B], [C|D]) :-
        C is A*A,
        squares(B, D).
  proof that sum_sq_evens/2 = ref_sum_sq_evens/2:
    proved for every input of the declared types:
      unfold sum_sq_evens_s1_test/1 (one clause, not recursive)
      unfold sum_sq_evens_s2_el/2 (one clause, not recursive)
      unfold sum_sq_evens_s3/2 (one clause, not recursive)
